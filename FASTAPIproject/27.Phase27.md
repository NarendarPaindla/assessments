## File: app/crud/cart.py

Add:

```python id="0n4jbn"
def update_cart_quantity(
    db: Session,
    user_id: int,
    cart_item_id: int,
    quantity: int
):

    cart = (
        db.query(Cart)
        .filter(
            Cart.user_id == user_id
        )
        .first()
    )

    if not cart:
        return False

    cart_item = (
        db.query(CartItem)
        .filter(
            CartItem.id == cart_item_id,
            CartItem.cart_id == cart.id
        )
        .first()
    )

    if not cart_item:
        return False

    if quantity <= 0:

        db.delete(cart_item)

        db.commit()

        return "deleted"

    cart_item.quantity = quantity

    db.commit()

    return "updated"
```

### Add Route

```python id="9g08cx"
@router.put(
    "/update/{cart_item_id}",
    response_model=
    CartMessageResponse
)
def update_quantity(
    cart_item_id: int,
    request:
    UpdateCartQuantityRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(
        get_current_user
    )
):

    result = update_cart_quantity(
        db=db,
        user_id=current_user.id,
        cart_item_id=cart_item_id,
        quantity=request.quantity
    )

    if not result:
        raise HTTPException(
            status_code=404,
            detail="Cart item not found"
        )

    if result == "deleted":

        return {
            "message":
            "Cart item removed"
        }

    return {
        "message":
        "Cart quantity updated"
    }
```
